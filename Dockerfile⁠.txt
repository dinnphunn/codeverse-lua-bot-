FROM ubuntu:22.04

RUN apt-get update && apt-get install -y curl ca-certificates unzip git build-essential

WORKDIR /app

# Cài đặt luvit/lit
RUN curl -L https://github.com/luvit/lit/raw/master/get-lit.sh | sh

COPY . .

# Tự động cài đặt các phụ thuộc nếu có file package.lua
RUN if [ -f "package.lua" ]; then ./lit install; fi

# Đảm bảo đường dẫn module tìm thấy thư viện trong thư mục deps
ENV LUA_PATH="/app/deps/?.lua;/app/deps/?/init.lua;./?.lua;./?/init.lua;/usr/local/share/luajit-2.1.0-beta3/?.lua;;"

CMD ["./luvit", "bot.lua"]
